C's mkdir

To do that in C you do:

#include <sys/type.h>
#include <sys/stat.h>
#include <errno.h>

if (mkdir("direcory", 0755) == -1) {
    if (errno == EEXIST) {
        printf("The directory already exists\n");
    } else {
        perror("mkdir");
    }
}
